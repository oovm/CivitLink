% ============================================

\label(use_flashlight)

\flash(0.5)

\p （手电筒的光照亮了走廊...）

\goto(explore_corridor)

% ============================================
% 分支：跟着黑崎
% ============================================

\label(follow_ren)

\p （相信黑崎同学吧。）

\p （他好像对我有点改观？）

\goto(explore_corridor)

% ============================================
% 分支：询问黑崎
% ============================================

\label(ask_ren_knowledge)

\p 黑崎同学，你怎么知道这个地方？

\r ...

\r 我...曾经在这里待过。

\p 什么意思？

\set(knows_ren_secret = true)

\p 实验...？

\r 以后再解释。先往前走。

\goto(explore_corridor)

% ============================================

\scene[old_building_hallway.png]

我们沿着走廊前进，两边是废弃的教室。

\flash(0.2)
\audio.play_se(strange_sound.wav)

\p 什么声音？

\choice
\case[hide_and_observe][+c, +t] 躲起来观察
\case[go_check][+r] 直接去查看
\case[send_ren] 让黑崎去查看
\end-choice

% ============================================
% 分支：躲起来观察
% ============================================

\label(hide_and_observe)

\p （先躲起来看看情况。）

\scene[old_building_shadow.png]

我们躲在一扇破旧的门后。

\r （小声）看来你被盯上了。

\char[unknown][hide]

\p （这到底是怎么回事...）

\goto(discover_room)

% ============================================
% 分支：直接去查看
% ============================================

\label(go_check)

\p （去看看是什么。）

\scene[old_building_intersection.png]

\set(clues_found += 1)

\p 这是...

\goto(discover_room)

% ============================================
% 分支：让黑崎去
% ============================================

\label(send_ren)

\p 黑崎同学，你能去看看吗？

\r ...切。

\r 在这里等着。

\char[ren][hide]

\pause(2.0)

\set(has_lab_access = true)

\p 一张门禁卡？

\r 旧校舍的地下实验室用的。

\goto(discover_room)

% ============================================
% 发现密室
% ============================================

\label(discover_room)

\r 这就是目的地。

\if(has_lab_access)
\goto(enter_with_card)
\else
\goto(locked_door)
\endif

% ============================================
% 锁住的门
% ============================================

\label(locked_door)

\p 门锁着...

\r 我们需要找到开门的方法。

\choice
\case[find_entrance][+c] 寻找其他入口

\end-choice

\audio.play_se(card_beep.wav)

% ============================================

% ============================================

\label(find_entrance)

\goto(enter_lab)

% 尝试撬锁
% ============================================

\label(try_lockpick)

% ============================================
% 找樱井帮忙
% ============================================

\label(get_sakura_help)

\char[sakura][worried, left]

% 进入实验室
% ============================================

\r 我是实验对象之一。

\label(promise_together)

\r ...

\r ...谢了。

\goto(chapter1_old_building_end)
